#!/bin/bash

apt -y update
apt -y install apache2 mc
systemctl start apache2
systemctl enable apache2
apt-get -y install libapache2-mod-ssl
apt -y install openssl
# Enable mod_ssl
a2enmod ssl

cat << EOF > /etc/ssl/site.info
[req]
prompt = no
distinguished_name = req_distinguished_name

[req_distinguished_name]
C = ua 
ST = LV
L = Lviv
O = IT-STEP
OU = Labs
CN = my.host.domain.com
emailAddress = mail@domain.com
EOF

# Set Up SSL Certificate
openssl req -x509 -nodes -days 365 -newkey rsa:2048 -keyout /etc/ssl/private/apache-selfsigned.key -out /etc/ssl/certs/apache-selfsigned.crt -config /etc/ssl/site.info
# Configure Apache to Use SSL
sed -i 's|SSLCertificateFile.*|SSLCertificateFile /etc/ssl/certs/apache-selfsigned.crt|g' /etc/apache2/sites-available/default-ssl.conf
sed -i 's|SSLCertificateKeyFile.*|SSLCertificateKeyFile /etc/ssl/private/apache-selfsigned.key|g' /etc/apache2/sites-available/default-ssl.conf
# Enable the SSL Site
a2ensite default-ssl
systemctl restart apache2
